cmake_minimum_required(VERSION 3.16)
project(cronicas_espaco VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Ativa ASan e UBSan" OFF)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
add_library(cronicas src/CelestialBody.cpp src/HttpClient.cpp src/BodyJsonParser.cpp src/SolarApiClient.cpp src/HashTable.cpp src/BodyCatalog.cpp)
target_include_directories(cronicas PUBLIC include)
target_link_libraries(cronicas PUBLIC CURL::libcurl nlohmann_json::nlohmann_json)
target_compile_options(cronicas PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(cronicas PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cronicas PUBLIC -fsanitize=address,undefined)
endif()
add_executable(missoes src/main.cpp)
target_link_libraries(missoes PRIVATE cronicas)
enable_testing()
add_executable(cronicas_tests tests/tests.cpp)
target_link_libraries(cronicas_tests PRIVATE cronicas)
target_compile_options(cronicas_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME cronicas_tests COMMAND cronicas_tests)
